cmake_minimum_required(VERSION 3.24)
project(wgs84_distance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(geographiclib_c STATIC vendor/geodesic.c)
target_include_directories(geographiclib_c PUBLIC vendor)
add_library(wgs84_distance STATIC src/wgs84_distance.cpp)
target_include_directories(wgs84_distance PUBLIC include)
target_link_libraries(wgs84_distance PUBLIC geographiclib_c)
if(MSVC)
  target_compile_options(geographiclib_c PRIVATE /O2 /fp:precise)
  target_compile_options(wgs84_distance PRIVATE /O2 /fp:precise /W4)
else()
  target_compile_options(geographiclib_c PRIVATE -O3 -fno-fast-math)
  target_compile_options(wgs84_distance PRIVATE -O3 -fno-fast-math -Wall -Wextra)
  target_link_libraries(geographiclib_c PUBLIC m)
endif()
find_package(OpenMP)
add_executable(wgs84_distance_cli src/main.cpp)
set_target_properties(wgs84_distance_cli PROPERTIES OUTPUT_NAME wgs84_distance)
target_link_libraries(wgs84_distance_cli PRIVATE wgs84_distance)
if(OpenMP_CXX_FOUND)
  target_link_libraries(wgs84_distance_cli PRIVATE OpenMP::OpenMP_CXX)
endif()
add_executable(distance_tests tests/distance_tests.cpp)
target_link_libraries(distance_tests PRIVATE wgs84_distance)
add_test(NAME distance_module COMMAND distance_tests)
foreach(name geodtest geodsigntest)
  add_executable(${name} vendor/${name}.c)
  target_link_libraries(${name} PRIVATE geographiclib_c)
  add_test(NAME upstream_${name} COMMAND ${name})
endforeach()
