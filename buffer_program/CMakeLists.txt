cmake_minimum_required(VERSION 3.24)
project(atlas_ellipsoidal_buffer LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BAFE_ENABLE_CUDA "Build the CUDA topology backend" ON)
if(BAFE_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  find_package(CUDAToolkit REQUIRED)
endif()
add_subdirectory(bafe)
add_subdirectory(distance)
add_executable(atlas_buffer src/main.cpp)
target_include_directories(atlas_buffer PRIVATE include bafe/src src)
target_link_libraries(atlas_buffer PRIVATE bafe::surface wgs84_distance)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(atlas_buffer PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(atlas_buffer PRIVATE /O2 /fp:precise /Zc:preprocessor /W4)
else()
  target_compile_options(atlas_buffer PRIVATE -O3 -fno-fast-math -Wall -Wextra)
endif()
add_test(NAME buffer_self_test COMMAND atlas_buffer --self-test)
